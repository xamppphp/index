3. Форма: радиокнопки с проверкой и сменой выбора
<form id="formID">
    Какая пасхалка круче?<br>
    <input type="radio" name="posx" id="r1"> 67<br>
    <input type="radio" name="posx" id="r2"> 69<br>
    <button type="button" onclick="rev()">Сменить выбор</button>
    <button type="button" onclick="prov()">Проверить</button>
</form>

<script>
// Функция для смены переключателей
function rev() {
    let r1 = document.getElementById('r1');
    let r2 = document.getElementById('r2');
    if (r1.checked) {
        r1.checked = false;
        r2.checked = true;
    } else if (r2.checked) {
        r2.checked = false;
        r1.checked = true;
    }
}
// Функция проверки выбора
function prov() {
    let r1 = document.getElementById('r1');
    let r2 = document.getElementById('r2');
    if (r1.checked) {
        alert('Ты красавчик');
    } else if (r2.checked) {
        alert('Ты даун?');
    } else {
        alert('Выбери вариант');
    }
}
</script>
// Кнопка "Сменить выбор" переключает радиокнопки, "Проверить" выводит сообщение
4. Выбор цвета: перевод на английский и окраска блока
<select id="color">
    <option>красный</option>
    <option>синий</option>
    <option>зеленый</option>
    <option>желтый</option>
    <option>фиолетовый</option>
</select>
<button onclick="go()">Применить</button>
<p id="block" style="padding:20px;border:1px solid #000">Цветной блок</p>
<p id="trans"></p>

<script>
function go() {
    let c = document.getElementById("color").value;
    let en;
    // switch - выбираем английское название цвета
    switch(c) {
        case "красный": en = "red"; break;
        case "синий": en = "blue"; break;
        case "зеленый": en = "green"; break;
        case "желтый": en = "yellow"; break;
        case "фиолетовый": en = "purple"; break;
        default: en = "black";
    }
    // Выводим перевод
    document.getElementById("trans").innerHTML = "Английский: " + en;
    // Меняем цвет фона блока
    document.getElementById("block").style.backgroundColor = en;
}
</script>
// При выборе цвета блок перекрашивается и показывается английское название

1. PHP. День недели по числу (передача через GET)

// isset() - проверяет, существует ли параметр, intval() - преобразует в целое число

    case 6: echo "Суббота"; break;
    case 7: echo "Воскресенье"; break;
    default: echo "Введите число от 1 до 7";
}
?>
// При day=1 выведет "Понедельник"

3. PHP. Подсчет слов, начинающихся на "Б"
<?php
$str = "Быстрый Боб бежит по Бульвару";
// strtolower() - переводит строку в нижний регистр для корректного сравнения
$str = strtolower($str);
// explode() - разбивает строку на массив по пробелу
$words = explode(" ", $str);
$count = 0;
foreach ($words as $word) {
    // mb_substr() - берет первый символ слова (для работы с кириллицей)
    if (mb_substr($word, 0, 1) === "б") {
        $count++;
    }
}
echo "Слов на 'Б': " . $count;
?>
// Результат: 3 (быстрый, боб, бульвару)

6. PHP. Числа от 10 до 60, кратные 7, их произведение
<?php
$a = 10;
$b = 61; // до 60 включительно
$product = 1;
for ($i = $a; $i < $b; $i++) {
    // % - оператор остатка от деления. Если остаток 0 - число кратно 7
    if ($i % 7 == 0) {
        $product *= $i;
    }
}
echo $product;
?>
// Кратные 7: 14, 21, 28, 35, 42, 49, 56. Их произведение = 14*21*28*35*42*49*56

9. PHP. Поиск числа, кратного 3 и 5 одновременно
<?php
$arr = [5, 2, 3, 4, 75, 6, 7, 8];
$len = count($arr); // количество элементов массива
for ($i = 0; $i < $len; $i++) {
    // && - логическое "И". Число должно делиться и на 3, и на 5
    if ($arr[$i] % 3 == 0 && $arr[$i] % 5 == 0) {
        echo "Найдено: " . $arr[$i];
        break; // прерываем цикл после нахождения
    }
}
?>
// Результат: 75 (делится и на 3, и на 5)

18. PHP. Обработка ошибки деления на ноль (try-catch)
<?php
function divide($a, $b) {
    if ($b == 0) {
        // throw - выбрасывает исключение
        throw new Exception("На ноль делить нельзя!");
    }
    return $a / $b;
}

try {
    echo divide(10, 0);
} catch (Exception $e) {
    // catch - перехватывает исключение
    echo "Ошибка: " . $e->getMessage();
}
?>
// Результат: "Ошибка: На ноль делить нельзя!"
